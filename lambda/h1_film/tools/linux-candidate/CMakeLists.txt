cmake_minimum_required(VERSION 3.20)
project(h1_linux_candidate C)
set(LINUX_SERVER_NET ON CACHE BOOL "Retain offline network film handlers")
set(LINUX_SERVER_SRC "${H1_SOURCE}" CACHE PATH "Archived upstream source")
add_subdirectory("${H1_SOURCE}/linux_server" server)
set(H1_SEQUENTIAL_SOURCE "${H1_GENERATED}/film_sequential.c")
file(READ "${H1_SOURCE}/source-revision.json" source_revision)
string(JSON source_commit GET "${source_revision}" git_commit)
if(source_commit STREQUAL "e78e4daa4b6d6b3e9ca290e3ebfe7a993ed558ad")
    include("${CMAKE_CURRENT_SOURCE_DIR}/beta4.cmake")
endif()
get_target_property(host_sources blam_server SOURCES)
list(FILTER host_sources EXCLUDE REGEX "(^|/)blampc_(film|pad_claim)\\.c$")
set_property(TARGET blam_server PROPERTY SOURCES "${host_sources}")
target_sources(blam_server PRIVATE "${H1_SEQUENTIAL_SOURCE}" "${H1_GENERATED}/export.c" "${H1_GENERATED}/candidate.c" "${CMAKE_CURRENT_SOURCE_DIR}/platform.c")
target_sources(blam_server PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/headless-extra.c")
target_include_directories(blam_server PRIVATE "${H1_SOURCE}/engine" "${H1_GENERATED}" "${CMAKE_CURRENT_SOURCE_DIR}/include" "/engines/beta3")
foreach(source IN ITEMS "${H1_SEQUENTIAL_SOURCE}" "${H1_GENERATED}/export.c")
    set_property(SOURCE "${source}" TARGET_DIRECTORY blam_server APPEND PROPERTY COMPILE_OPTIONS
        -include linux_server_prelude.h -include blampc_prelude.h)
endforeach()
set_property(SOURCE "${H1_GENERATED}/export.c" TARGET_DIRECTORY blam_server APPEND PROPERTY COMPILE_OPTIONS -fbuiltin)
set_property(SOURCE "${H1_GENERATED}/export.c" TARGET_DIRECTORY blam_server APPEND PROPERTY COMPILE_DEFINITIONS WC_ERR_INVALID_CHARS=0x80 sprintf_s=snprintf)
set_property(SOURCE "${H1_GENERATED}/export.c" "${H1_SOURCE}/host/blampc_net.c" TARGET_DIRECTORY blam_server
    APPEND PROPERTY COMPILE_DEFINITIONS _POSIX_C_SOURCE=200809L)
set_property(SOURCE "${H1_SEQUENTIAL_SOURCE}" "${H1_SOURCE}/host/blampc_net_rec.c" "${H1_SOURCE}/host/blampc_net_trace.c" "${H1_SOURCE}/host/blampc_picture_probe.c" TARGET_DIRECTORY blam_server
    APPEND PROPERTY COMPILE_OPTIONS -include candidate-platform.h)
set_property(SOURCE "${H1_SOURCE}/host/main.c" TARGET_DIRECTORY blam_server APPEND PROPERTY COMPILE_DEFINITIONS
    blampc_state_dump=hs_export_tick blampc_spawn_odds_tick=hs_unused_spawn_odds_tick)
set_property(SOURCE "${H1_SOURCE}/engine/blam/object_cause_damage.c" TARGET_DIRECTORY blam_server APPEND PROPERTY COMPILE_DEFINITIONS
    blampc_pgcr_stats_note_damage=hs_note_damage)
set_property(SOURCE "${H1_SOURCE}/linux_server/platform/linux_film_input.c" TARGET_DIRECTORY blam_server APPEND PROPERTY COMPILE_DEFINITIONS
    blampc_menu_build_label=hs_original_build_label)
set_property(SOURCE "${H1_SOURCE}/linux_server/platform/linux_film_input.c" TARGET_DIRECTORY blam_server APPEND PROPERTY COMPILE_OPTIONS -include assert.h)
set_target_properties(blam_server PROPERTIES OUTPUT_NAME h1-film-linux)
option(H1_OPTIMIZE "Optimize the offline native engine without relaxed floating point" OFF)
set(H1_LTO "thin" CACHE STRING "Native link-time optimizer: thin or full")
set(H1_ARCH "" CACHE STRING "Optional native CPU instruction-set target")
set(H1_PGO_GENERATE "" CACHE PATH "Instrumentation output directory")
set(H1_PGO_USE "" CACHE FILEPATH "Merged instrumentation profile")
if(H1_PGO_GENERATE AND H1_PGO_USE)
    message(FATAL_ERROR "Choose profile generation or consumption, not both")
endif()
if(H1_OPTIMIZE)
    set_property(SOURCE "${H1_GENERATED}/export.c" TARGET_DIRECTORY blam_server APPEND PROPERTY COMPILE_DEFINITIONS H1_BUFFERED_EXPORT=1)
    target_compile_options(blam_server PRIVATE -O3 "-flto=${H1_LTO}" -fno-pic -fno-pie -fbuiltin -fno-builtin-fmod)
    target_link_options(blam_server PRIVATE "-flto=${H1_LTO}")
    if(H1_ARCH)
        target_compile_options(blam_server PRIVATE "-march=${H1_ARCH}")
    endif()
    set_property(SOURCE "${H1_SOURCE}/linux_server/platform/linux_win32.c" "${H1_SOURCE}/linux_server/platform/linux_sync.c"
        TARGET_DIRECTORY blam_server APPEND PROPERTY COMPILE_DEFINITIONS usleep=hs_usleep)
endif()
if(H1_PGO_GENERATE)
    target_compile_options(blam_server PRIVATE "-fprofile-generate=${H1_PGO_GENERATE}")
    target_link_options(blam_server PRIVATE "-fprofile-generate=${H1_PGO_GENERATE}")
elseif(H1_PGO_USE)
    target_compile_options(blam_server PRIVATE "-fprofile-use=${H1_PGO_USE}")
    target_link_options(blam_server PRIVATE "-fprofile-use=${H1_PGO_USE}")
endif()
option(H1_PROFILE "Instrument the offline candidate with gprof" OFF)
if(H1_PROFILE)
    target_compile_options(blam_server PRIVATE -pg)
    target_link_options(blam_server PRIVATE -pg)
endif()
